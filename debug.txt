import { useEffect, useState } from 'react'
import AuthFlow from './components/AuthFlow.jsx'
import Onboarding from './components/Onboarding.jsx'
import Today from './pages/Today.jsx'
import Wardrobe from './pages/Wardrobe.jsx'
import Explore from './pages/Explore.jsx'
import Preferences from './pages/Preferences.jsx'
import MyProfile from './pages/MyProfile.jsx'
import { loadFromStorage, saveToStorage, storageKeys } from './utils/storage.js'
import { createDefaultProfile, normalizeProfile } from './utils/bodyProfile.js'
import { generateOutfits } from './engine/index.js'
import guestWardrobeSeed from './data/guestWardrobe.json'
import { exportWardrobePayload, importWardrobeRecords } from './utils/wardrobeStorage.js'
import { apiRequest, authApi, setUnauthorizedHandler } from './utils/api.js'
import { loadUserSession, syncUserData } from './utils/syncData.js'
import { outfitTemplates } from './data/outfitTemplates.js'
import { syncQueuedOperations } from './utils/offlineSync.js'

const demoAccount = { id: 'demo-guest', name: 'Rael', email: 'guest@fitzii.local', guest: true }
const demoPreferences = {
  styles: ['minimal', 'classic'],
  colors: ['ivory', 'navy', 'olive'],
  materials: ['cotton', 'linen', 'denim'],
  formalityBias: 3,
}

const defaultPreferences = demoPreferences

const navItems = [
  ['today', 'Today'],
  ['wardrobe', 'Wardrobe'],
  ['explore', 'Explore'],
  ['profile', 'My Profile'],
  ['preferences', 'Preferences'],
]

const navIcons = {
  today: 'M3 10.5 12 3l9 7.5M5.5 9v11h13V9M9 20v-6h6v6',
  wardrobe: 'M4 4h6v6H4zM14 4h6v6h-6zM4 14h6v6H4zM14 14h6v6h-6z',
  explore: 'm12 3 1.9 5.8L20 11l-6.1 2.2L12 19l-1.9-5.8L4 11l6.1-2.2L12 3ZM19 15l1 2.5 2.5 1-2.5 1-1 2.5-1-2.5-2.5-1 2.5-1 1-2.5Z',
  profile: 'M20 21a8 8 0 0 0-16 0M12 13a5 5 0 1 0 0-10 5 5 0 0 0 0 10Z',
  preferences: 'M4 6h16M4 12h16M4 18h16M9 4v4m7 2v4m-5 2v4',
}

function NavIcon({ name }) {
  return <svg aria-hidden="true" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="1.7" strokeLinecap="round" strokeLinejoin="round" className="h-5 w-5">{name === 'explore' ? <path d={navIcons[name]} fill="currentColor" stroke="none" /> : <path d={navIcons[name]} />}</svg>
}

function App() {
  const [garments, setGarments] = useState([])
  const [page, setPage] = useState('today')
  const [context, setContext] = useState('casual')
  const [aesthetic, setAesthetic] = useState('minimal')
  const [preferences, setPreferences] = useState(defaultPreferences)
  const [savedOutfits, setSavedOutfits] = useState([])
  const [history, setHistory] = useState([])
  const [selectedGarment, setSelectedGarment] = useState(null)
  const [profile, setProfile] = useState(createDefaultProfile())
  const [account, setAccount] = useState(() => loadFromStorage(storageKeys.account, null))
  const [stage, setStage] = useState(() => {
    const activeAccount = loadFromStorage(storageKeys.account, null)
    const guestMode = loadFromStorage('fitzii.guest', false)
    return activeAccount || guestMode ? 'app' : 'auth'
  })
  const [showTutorial, setShowTutorial] = useState(false)
  const [apiError, setApiError] = useState('')
  const [isOffline, setIsOffline] = useState(typeof navigator === 'undefined' ? false : !navigator.onLine)
  const [isLoading, setIsLoading] = useState(true)
  const [recommendedOutfits, setRecommendedOutfits] = useState([])
  const [aestheticOutfits, setAestheticOutfits] = useState([])
  const [template, setTemplate] = useState(null)

  const hydrateAuthenticatedSession = async (activeAccount) => {
    if (!activeAccount) return

    setAccount(activeAccount)
    saveToStorage(storageKeys.account, activeAccount)

    try {
      const syncResult = await loadUserSession(activeAccount)
      const wardrobe = Array.isArray(syncResult?.wardrobe) ? syncResult.wardrobe : loadFromStorage(storageKeys.wardrobe, guestWardrobeSeed)
      const profile = normalizeProfile(syncResult?.profile && Object.keys(syncResult.profile).length ? syncResult.profile : loadFromStorage(storageKeys.profile, createDefaultProfile()))
      const preferences = syncResult?.preferences && Object.keys(syncResult.preferences).length ? syncResult.preferences : loadFromStorage(storageKeys.preferences, defaultPreferences)

      setGarments(wardrobe)
      setProfile(profile)
      setPreferences(preferences)
      saveToStorage(storageKeys.wardrobe, wardrobe)
      saveToStorage(storageKeys.profile, profile)
      saveToStorage(storageKeys.preferences, preferences)
    } catch (error) {
      console.warn('Could not hydrate backend session data, keeping local data as fallback.', error)
      const localWardrobe = loadFromStorage(storageKeys.wardrobe, guestWardrobeSeed)
      const localProfile = loadFromStorage(storageKeys.profile, createDefaultProfile())
      const localPreferences = loadFromStorage(storageKeys.preferences, defaultPreferences)
      setGarments(localWardrobe)
      setProfile(localProfile)
      setPreferences(localPreferences)
      saveToStorage(storageKeys.wardrobe, localWardrobe)
      saveToStorage(storageKeys.profile, localProfile)
      saveToStorage(storageKeys.preferences, localPreferences)
    }
  }

  useEffect(() => {
    setUnauthorizedHandler(() => {
      setAccount(null)
      saveToStorage(storageKeys.account, null)
      setStage('auth')
    })
  }, [])

  useEffect(() => {
    const storedSaved = loadFromStorage(storageKeys.outfits, [])
    const existingHistory = loadFromStorage(storageKeys.history, [])
    const activeAccount = loadFromStorage(storageKeys.account, null)
    const localWardrobe = loadFromStorage(storageKeys.wardrobe, guestWardrobeSeed)
    const localProfile = loadFromStorage(storageKeys.profile, createDefaultProfile())
    const localPreferences = loadFromStorage(storageKeys.preferences, defaultPreferences)

    setAccount(activeAccount)
    setSavedOutfits(storedSaved)
    setGarments(localWardrobe)
    setProfile(localProfile)
    setPreferences(localPreferences)
    setHistory(existingHistory)
    setApiError('')

    const bootstrap = async () => {
      try {
        const currentUser = activeAccount ? activeAccount : await authApi.me()
        if (!currentUser) {
          setStage('auth')
          setIsLoading(false)
          return
        }

        const nextAccount = {
          id: currentUser.id,
          name: currentUser.displayName || currentUser.name || currentUser.username,
          username: currentUser.username,
          password: activeAccount?.password || '',
        }

        saveToStorage(storageKeys.account, nextAccount)
        setAccount(nextAccount)
        setStage('app')
        const remoteData = await syncUserData(nextAccount, {
          wardrobe: localWardrobe,
          profile: localProfile,
          preferences: localPreferences,
          history: existingHistory,
        })

        const wardrobe = Array.isArray(remoteData?.wardrobe) ? remoteData.wardrobe : localWardrobe
        const profile = normalizeProfile(remoteData?.profile && Object.keys(remoteData.profile).length ? remoteData.profile : localProfile)
        const preferences = remoteData?.preferences && Object.keys(remoteData.preferences).length ? remoteData.preferences : localPreferences

        setGarments(wardrobe)
        setProfile(profile)
        setPreferences(preferences)
        setHistory(Array.isArray(remoteData?.history) ? remoteData.history : existingHistory)
        saveToStorage(storageKeys.wardrobe, wardrobe)
        saveToStorage(storageKeys.profile, profile)
        saveToStorage(storageKeys.preferences, preferences)
        saveToStorage(storageKeys.history, Array.isArray(remoteData?.history) ? remoteData.history : existingHistory)
      } catch (error) {
        console.warn('App bootstrap could not reach the backend; using local data instead.', error)
        setStage(activeAccount ? 'app' : 'auth')
      } finally {
        setIsLoading(false)
      }
    }

    bootstrap()
  }, [])

  useEffect(() => {
    const handleOnline = () => {
      setIsOffline(false)
    }
    const handleOffline = () => setIsOffline(true)

    window.addEventListener('online', handleOnline)
    window.addEventListener('offline', handleOffline)
    setIsOffline(!navigator.onLine)

    return () => {
      window.removeEventListener('online', handleOnline)
      window.removeEventListener('offline', handleOffline)
    }
  }, [])

  useEffect(() => {
    saveToStorage(storageKeys.outfits, savedOutfits)
  }, [savedOutfits])

  useEffect(() => {
    if (!garments.length) return
    const base = { garments, context, history, profile, limit: 6, template }
    const general = generateOutfits({ ...base, preferences: { ...preferences, aesthetic } })
    const favorite = generateOutfits({ ...base, preferences: { ...preferences, aesthetic, styles: [aesthetic] } })
    setRecommendedOutfits(general)
    setAestheticOutfits(favorite)
  }, [aesthetic, context, garments, preferences, profile, history, template])

  useEffect(() => {
    if (!isOffline && account) syncQueuedOperations(apiRequest).catch(() => { })
  }, [isOffline, account])

  const persistPreferences = (next) => {
    setPreferences(next)
    saveToStorage(storageKeys.preferences, next)
    apiRequest('/preferences', { method: 'PUT', body: JSON.stringify(next) }).catch(() => { })
    setApiError('')
  }

  const handleSaveOutfit = (entry) => {
    const newSaved = [
      {
        id: `saved-${Date.now()}`,
        createdAt: new Date().toISOString(),
        outfit: entry.outfit,
        score: entry.score,
      },
      ...savedOutfits,
    ].slice(0, 8)

    setSavedOutfits(newSaved)
    const timestamp = Date.now()
    setHistory((current) => [{ items: entry.outfit, timestamp }, ...current].slice(0, 12))
    saveToStorage(storageKeys.history, [{ items: entry.outfit, timestamp }, ...history].slice(0, 12))
    apiRequest('/history', { method: 'POST', body: JSON.stringify({ id: `saved-${timestamp}`, garmentIds: entry.outfit.map((item) => item.id), context, score: entry.score }) }).catch(() => { })
  }

  const handleSignUp = async ({ name, username, password }) => {
    const cleanedUsername = String(username || name || '').trim()
    const displayName = String(name || '').trim() || cleanedUsername

    try {
      const remoteUser = await authApi.register({
        username: cleanedUsername,
        password,
        name: displayName,
      })

      const remoteAccount = {
        id: remoteUser.id,
        name: remoteUser.displayName || displayName,
        username: remoteUser.username || cleanedUsername,
        password,
      }

      const saved = loadFromStorage(storageKeys.accounts, [])
      const nextAccounts = [...(Array.isArray(saved) ? saved : []), remoteAccount]
      saveToStorage(storageKeys.accounts, nextAccounts)
      saveToStorage(storageKeys.account, remoteAccount)
      setAccount(remoteAccount)
      await hydrateAuthenticatedSession(remoteAccount)
      setStage('onboarding')
      return true
    } catch (error) {
      console.warn('Backend account creation failed, falling back to local storage.', error)
      if (Number(error?.status) > 0) throw error
    }

    const localAccount = {
      id: crypto.randomUUID?.() || `account-${Date.now()}`,
      name: displayName,
      username: cleanedUsername,
      password,
    }
    const saved = loadFromStorage(storageKeys.accounts, [])
    const nextAccounts = [...(Array.isArray(saved) ? saved : []), localAccount]
    saveToStorage(storageKeys.accounts, nextAccounts)
    saveToStorage(storageKeys.account, localAccount)
    setAccount(localAccount)
    setStage('onboarding')
    return true
  }

  const handleLogin = async (identifier, password) => {
    const trimmedIdentifier = String(identifier || '').trim()
    const normalizedPassword = String(password || '')

    try {
      const remoteUser = await authApi.login({
        username: trimmedIdentifier,
        password: normalizedPassword,
      })

      const nextAccount = {
        id: remoteUser.id,
        name: remoteUser.displayName || remoteUser.name || trimmedIdentifier,
        username: remoteUser.username || trimmedIdentifier,
        password: normalizedPassword,
      }

      saveToStorage(storageKeys.account, nextAccount)
      setAccount(nextAccount)
      await hydrateAuthenticatedSession(nextAccount)
      setStage('app')
      setPage('today')
      return true
    } catch (error) {
      console.warn('Backend login failed, falling back to local storage only if a matching offline record exists.', error)
    }

    const saved = loadFromStorage(storageKeys.accounts, [])
    const found = (Array.isArray(saved) ? saved : []).find((account) => {
      const accountUsername = String(account.username || account.name || account.email || '').trim().toLowerCase()
      const accountName = String(account.name || '').trim().toLowerCase()
      const accountEmail = String(account.email || '').trim().toLowerCase()
      const storedPassword = String(account.password || '')

      const matchesIdentifier = accountUsername === trimmedIdentifier.toLowerCase() || accountName === trimmedIdentifier.toLowerCase() || accountEmail === trimmedIdentifier.toLowerCase()
      const matchesPassword = !storedPassword || storedPassword === normalizedPassword

      return matchesIdentifier && matchesPassword
    })

    if (!found) return false

    const nextAccount = { ...found, password: found.password || normalizedPassword }
    saveToStorage(storageKeys.account, nextAccount)
    setAccount(nextAccount)
    setStage('app')
    setPage('today')
    return true
  }

  const handleResetPassword = ({ username, password }) => {
    const saved = loadFromStorage(storageKeys.accounts, [])
    const normalizedUsername = String(username || '').trim().toLowerCase()
    const nextAccounts = (Array.isArray(saved) ? saved : []).map((account) => {
      const accountUsername = String(account.username || account.name || account.email || '').trim().toLowerCase()
      if (accountUsername === normalizedUsername) {
        return { ...account, password }
      }
      return account
    })

    if (!nextAccounts.some((account) => {
      const accountUsername = String(account.username || account.name || account.email || '').trim().toLowerCase()
      return accountUsername === normalizedUsername
    })) {
      return false
    }

    saveToStorage(storageKeys.accounts, nextAccounts)
    const active = loadFromStorage(storageKeys.account, null)
    if (active) {
      const activeUsername = String(active.username || active.name || active.email || '').trim().toLowerCase()
      if (activeUsername === normalizedUsername) {
        saveToStorage(storageKeys.account, { ...active, password })
      }
    }
    return true
  }

  const handleGuest = () => {
    saveToStorage('fitzii.guest', true)
    setStage('app')
    setPage('today')
  }

  const handleOnboardingFinish = ({ aesthetic: nextAesthetic, preferences: nextPrefs }) => {
    setAesthetic(nextAesthetic)
    persistPreferences(nextPrefs)
    setStage('app')
    setShowTutorial(true)
    setPage('today')
  }

  const handleSignOut = async () => {
    try {
      await authApi.logout()
    } catch (error) {
      console.warn('Logout from the server failed, continuing with local session reset.', error)
    }

    saveToStorage(storageKeys.account, null)
    saveToStorage('fitzii.guest', false)
    setAccount(null)
    setStage('auth')
  }

  const handleOpenApp = () => {
    saveToStorage(storageKeys.introSeen, true)
    setPage('today')
  }

  const handleAddGarment = async (garment) => {
    const normalized = { status: 'available', ...garment }
    const saved = { ...normalized, id: normalized.id || `g-${Date.now()}` }
    setGarments((current) => {
      const next = [saved, ...current]
      saveToStorage(storageKeys.wardrobe, next)
      return next
    })
    apiRequest('/wardrobe', { method: 'POST', body: JSON.stringify(saved) }).catch(() => { })
    setApiError('')
    return saved
  }

  const handleUpdateGarment = async (garment) => {
    const saved = { ...garment, status: garment.status || 'available' }
    setGarments((current) => {
      const next = current.map((item) => item.id === saved.id ? saved : item)
      saveToStorage(storageKeys.wardrobe, next)
      return next
    })
    apiRequest(`/wardrobe/${encodeURIComponent(saved.id)}`, { method: 'PATCH', body: JSON.stringify(saved) }).catch(() => { })
    setApiError('')
    return saved
  }

  const handleImageUpdate = (saved) => {
    if (!saved || !saved.id) return
    setGarments((current) => {
      const next = current.map((item) => item.id === saved.id ? { ...item, ...saved } : item)
      saveToStorage(storageKeys.wardrobe, next)
      return next
    })
  }

  const handleDeleteGarment = async (garmentId) => {
    setGarments((current) => {
      const next = current.filter((item) => item.id !== garmentId)
      saveToStorage(storageKeys.wardrobe, next)
      return next
    })
    apiRequest(`/wardrobe/${encodeURIComponent(garmentId)}`, { method: 'DELETE' }).catch(() => { })
    setApiError('')
    setSavedOutfits((current) => current.filter((entry) => !entry.outfit?.some((item) => item.id === garmentId)))
  }

  const toggleGarmentAvailability = (garmentId, available = true) => {
    const item = garments.find((entry) => entry.id === garmentId)
    if (item) handleUpdateGarment({ ...item, status: available ? 'available' : 'unavailable' })
  }

  const handleExportWardrobe = () => {
    const payload = exportWardrobePayload(garments)
    const blob = new Blob([JSON.stringify(payload, null, 2)], { type: 'application/json' })
    const url = URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = url
    link.download = 'wardrobe-export.json'
    document.body.appendChild(link)
    link.click()
    link.remove()
    URL.revokeObjectURL(url)
  }

  const handleImportWardrobe = async (event) => {
    const [file] = event.target.files || []
    if (!file) return

    try {
      const text = await file.text()
      const parsed = JSON.parse(text)
      const imported = Array.isArray(parsed?.garments) ? parsed.garments : Array.isArray(parsed) ? parsed : []
      if (!imported.length) {
        window.alert('Import failed: the selected file is not a valid wardrobe export.')
        return
      }

      const merge = window.confirm(`Import ${imported.length} garments? Existing records with matching IDs will be kept.`)
      if (!merge) return

      const currentIds = new Set(garments.map((item) => item.id))
      const next = [...garments]
      for (const garment of imported) {
        if (!currentIds.has(garment.id)) {
          next.push({ ...garment, status: garment.status || 'available' })
        }
      }

      setGarments(next)
      saveToStorage(storageKeys.wardrobe, next)
    } catch (error) {
      console.error(error)
      window.alert('Import failed: the selected file is not a valid wardrobe export.')
    } finally {
      event.target.value = ''
    }
  }

  const notice = 'rounded-xl border px-4 py-3 text-sm'
  const savedAccounts = loadFromStorage(storageKeys.accounts, [])

  if (stage === 'auth') {
    return <AuthFlow accounts={savedAccounts} onSignUp={handleSignUp} onLogin={handleLogin} onResetPassword={handleResetPassword} onGuest={handleGuest} />
  }

  if (stage === 'onboarding') {
    return (
      <Onboarding
        name={account?.name || 'there'}
        initialPreferences={preferences}
        initialAesthetic={aesthetic}
        onFinish={handleOnboardingFinish}
        onSkip={() => {
          setStage('app')
          setPage('today')
          setShowTutorial(true)
        }}
      />
    )
  }

  return (
    <div className="min-h-screen lg:grid lg:grid-cols-[272px_1fr]">
      <aside className="hidden flex-col gap-5 border-r border-[#e8e2dc] bg-[#fffdfa] p-5 lg:sticky lg:top-0 lg:flex lg:h-screen lg:overflow-y-auto">
        <div>
          <p className="text-sm font-medium text-accent">Fitzii</p>
          <h1 className="text-2xl font-bold leading-tight tracking-tight">Dress with more ease</h1>
        </div>

        <div className="rounded-xl bg-accent-soft p-4">
          <span className="text-xs font-medium text-accent">Account</span>
          <strong className="block text-base">{account?.name || 'Guest'}</strong>
          <small className="text-slate-600">{account?.email || 'Local profile'}</small>
        </div>

        <nav className="flex gap-2 overflow-x-auto lg:flex-col lg:overflow-visible" aria-label="Main">
          {navItems.map(([key, label]) => (
            <button
              key={key}
              type="button"
              onClick={() => setPage(key)}
              aria-current={page === key ? 'page' : undefined}
              className={`shrink-0 rounded-lg px-4 py-2.5 text-left text-sm font-semibold transition focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-accent ${page === key ? 'bg-ink text-white' : 'text-slate-700 hover:bg-slate-100'
                }`}
            >
              {label}
            </button>
          ))}
        </nav>

        <div className="flex flex-col gap-2">
          <button type="button" className="rounded-lg bg-accent-soft px-4 py-2.5 text-sm font-semibold text-accent transition hover:bg-violet-200" onClick={account ? handleSignOut : () => setStage('auth')}>
            {account ? 'Log out' : 'Create account'}
          </button>
        </div>

        <div className="card !p-4 lg:mt-auto">
          <h3 className="text-sm font-medium text-slate-500">Saved picks</h3>
          <strong className="block text-3xl font-bold">{savedOutfits.length}</strong>
          <span className="text-sm text-slate-500">Curated outfits</span>
        </div>

        {selectedGarment ? (
          <div className="card !p-4">
            <h3 className="text-sm font-medium text-slate-500">Selected item</h3>
            <strong className="block">{selectedGarment.name}</strong>
            <span className="text-sm capitalize text-slate-500">
              {selectedGarment.visual_attributes?.primary_colors?.[0] || selectedGarment.color || 'unknown'} · {selectedGarment.style_associations?.possible_aesthetics?.[0] || selectedGarment.style || 'varied'}
            </span>
          </div>
        ) : null}
      </aside>

      <header className="sticky top-0 z-30 flex items-center justify-between border-b border-[#eee8e2] bg-[#f8f6f2]/95 px-4 py-3 backdrop-blur lg:hidden">
        <button type="button" onClick={() => setPage('today')} className="flex min-h-11 items-center gap-2 text-left" aria-label="Fitzii home">
          <span className="grid h-9 w-9 place-items-center rounded-2xl bg-accent text-sm font-bold text-white">f.</span>
          <span><strong className="block text-sm leading-4">Fitzii</strong><small className="text-[11px] text-slate-500">Your personal stylist</small></span>
        </button>
        <div className="flex items-center gap-2">
          {isOffline ? <span className="rounded-full bg-amber-100 px-2.5 py-1 text-[11px] font-semibold text-amber-900">Offline</span> : null}
          <button type="button" onClick={account ? handleSignOut : () => setStage('auth')} className="grid h-10 w-10 place-items-center rounded-full border border-[#e8e2dc] bg-white text-sm font-bold text-accent" aria-label={account ? 'Log out' : 'Create account'}>
            {(account?.name || 'G').trim().charAt(0).toUpperCase()}
          </button>
        </div>
      </header>

      <main className="page-bottom-safe mx-auto w-full max-w-6xl space-y-5 px-4 pt-4 sm:px-8 sm:pt-8 lg:space-y-6 lg:px-8 lg:pb-8">
        {!account ? (
          <div className="flex items-center justify-between gap-3 rounded-2xl border border-violet-200 bg-violet-50 px-4 py-3 text-sm text-violet-900">
            <span>Guest mode is active. Your data stays in this browser.</span>
            <button type="button" className="rounded-full bg-violet-600 px-3 py-1.5 font-semibold text-white" onClick={() => setStage('auth')}>Create account</button>
          </div>
        ) : null}
        {showTutorial ? (
          <div className="rounded-2xl border border-slate-200 bg-white p-4 shadow-sm">
            <p className="text-sm font-semibold text-slate-900">Quick start</p>
            <div className="mt-3 grid gap-3 md:grid-cols-3">
              {['Pick your context', 'Tune your aesthetic', 'Save your favorite outfits'].map((item, index) => (
                <div key={item} className="rounded-xl border border-slate-200 bg-slate-50 px-3 py-2 text-sm text-slate-700">
                  <span className="mr-2 inline-flex h-6 w-6 items-center justify-center rounded-full bg-violet-100 text-xs font-bold text-violet-700">{index + 1}</span>
                  {item}
                </div>
              ))}
            </div>
          </div>
        ) : null}
        {apiError ? <div className={`${notice} border-rose-200 bg-rose-50 text-rose-800`} role="alert">{apiError}</div> : null}
        {isOffline ? <div className={`${notice} border-amber-200 bg-amber-50 text-amber-900`} role="status">Offline mode is on. Your changes will sync automatically when the connection returns.</div> : null}
        {isLoading ? <div className={`${notice} border-slate-200 bg-white text-slate-600`} role="status">Connecting to your wardrobe…</div> : null}

        {page === 'today' ? (
          <>
            <div className="relative overflow-hidden rounded-[26px] bg-[#352631] px-5 py-5 text-white shadow-[0_15px_35px_rgba(53,38,49,0.14)] sm:px-7 sm:py-7">
              <div className="pointer-events-none absolute -right-12 -top-16 h-48 w-48 rounded-full border border-white/10" />
              <div className="pointer-events-none absolute -right-2 -top-7 h-32 w-32 rounded-full border border-white/10" />

function HelpPopover({ title, text }) {

        aria-expanded={isOpen}
        aria-haspopup="dialog"

      >

        </div>
      ) : null}
    </div>
  )

  top: ['tee', 'shirt', 'knit', 'sweater', 'blouse', 'polo', 'tank', 'dress'],

  top: ['basic', 'button-up', 'oversized', 'fitted', 'linen', 'striped', 'graphic', 'cotton'],
  bottom: ['straight', 'wide-leg', 'slim', 'cargo', 'flare', 'linen', 'denim'],
  outerwear: ['structured', 'relaxed', 'cropped', 'oversized', 'tailored', 'layering'],
  accessory: ['leather', 'canvas', 'wool', 'metal', 'structured', 'minimal'],
  shoe: ['low-profile', 'chunky', 'sleek', 'ankle-height', 'workwear', 'dress'],

const blankDraft = {
  id: '',
  name: '',
  category: 'top',
  type: 'tee',
  subtype: 'basic',
  brand: '',
  size: 'M',
  sizeSystem: 'us',
  status: 'available',
  color: 'ivory',
  fit: 'regular',
  style: 'minimal',
  aesthetic: 'clean',
  pattern: 'solid',
  material: 'cotton',
  visual_attributes: { primary_colors: ['ivory'], secondary_colors: [], pattern: 'solid', texture: 'smooth' },
  attributes: { formality_estimate: 0.5, visual_weight_estimate: 0.5, seasonal_suitability: [], functional_tags: [] },
  fit_and_proportions: { apparent_fit: 'regular', silhouette: 'straight', relative_volume: 'medium', relative_length: 'regular' },
}

function buildGarmentFromDraft(draft) {
  return {
    id: draft.id || `g-${Date.now()}`,
    name: draft.name || 'Untitled garment',
    category: draft.category || 'top',
    type: draft.type || 'tee',
    subtype: draft.subtype || '',
    brand: draft.brand || '',
    size: draft.size || '',
    sizeSystem: draft.sizeSystem || 'us',
    status: draft.status || 'available',
    color: draft.color || 'unknown',
    hex: draft.hex || '#d9d9d9',
    style: draft.style || 'minimal',
    aesthetic: draft.aesthetic || 'clean',
    pattern: draft.pattern || 'solid',
    material: draft.material || 'unknown',
    fit: draft.fit || 'regular',
    silhouette: draft.fit_and_proportions?.silhouette || 'straight',
    season: draft.season || 'all',
    visual_attributes: {
      primary_colors: draft.visual_attributes?.primary_colors || [draft.color || 'unknown'],
      secondary_colors: draft.visual_attributes?.secondary_colors || [],
      pattern: draft.visual_attributes?.pattern || 'solid',
      texture: draft.visual_attributes?.texture || 'smooth',
      visible_details: draft.visible_details || '',
    },
    attributes: {
      formality_estimate: Number(draft.attributes?.formality_estimate ?? 0.5),
      visual_weight_estimate: Number(draft.attributes?.visual_weight_estimate ?? 0.5),
      seasonal_suitability: draft.attributes?.seasonal_suitability || [],
      functional_tags: draft.attributes?.functional_tags || [],
    },
    fit_and_proportions: {
      apparent_fit: draft.fit_and_proportions?.apparent_fit || draft.fit || 'regular',
      silhouette: draft.fit_and_proportions?.silhouette || 'straight',
      relative_volume: draft.fit_and_proportions?.relative_volume || 'medium',
      relative_length: draft.fit_and_proportions?.relative_length || 'regular',
      optional_measurements: draft.fit_and_proportions?.optional_measurements || {},
    },
  }
}

function Field({ label, children, className = '' }) {
  return (
    <label className={`field ${className}`}>
      {label}
      {children}
    </label>
  )
}

export default function Wardrobe({ garments, onSelectGarment, onAddGarment, onUpdateGarment, onImageUpdate, onDeleteGarment, onToggleGarmentAvailability, onExportWardrobe, onImportWardrobe }) {
  const [isFormOpen, setIsFormOpen] = useState(false)
  const [mode, setMode] = useState('quick')
  const [draft, setDraft] = useState(blankDraft)
  const [search, setSearch] = useState('')
  const [statusFilter, setStatusFilter] = useState('all')
  const [categoryFilter, setCategoryFilter] = useState('all')
  const [editingId, setEditingId] = useState(null)
  const [imageFile, setImageFile] = useState(null)

  const visibleGarments = useMemo(() => {
    return garments.filter((garment) => {
      const matchesSearch = !search || garment.name?.toLowerCase().includes(search.toLowerCase())
      const matchesStatus = statusFilter === 'all' || garment.status === statusFilter
      const matchesCategory = categoryFilter === 'all' || garment.category === categoryFilter
      return matchesSearch && matchesStatus && matchesCategory
    })
  }, [categoryFilter, garments, search, statusFilter])

  const closeForm = () => {
    setIsFormOpen(false)
    setEditingId(null)
    setImageFile(null)
    setDraft(blankDraft)
  }

  const openNew = () => {
    setDraft(blankDraft)
    setEditingId(null)
    setImageFile(null)
    setMode('quick')
    setIsFormOpen(true)
  }

  const handleSubmit = async () => {
    const garment = buildGarmentFromDraft(draft)
    const file = imageFile
    const targetId = editingId
    closeForm()

    if (targetId) {
      await onUpdateGarment?.({ ...garment, id: targetId })
      if (file) onImageUpdate?.(await wardrobeApi.upload(targetId, file))
    } else {
      const saved = await onAddGarment?.(garment)
      if (file && saved) onImageUpdate?.(await wardrobeApi.upload(saved.id, file))
    }
  }

  const handleDraftChange = (field, value) => setDraft((current) => ({ ...current, [field]: value }))
  const text = (field) => ({ className: 'input', value: draft[field] ?? '', onChange: (e) => handleDraftChange(field, e.target.value) })
  const score = (field) => ({
    className: 'input',
    type: 'number', min: '0', max: '1', step: '0.1',
    value: draft.attributes?.[field] ?? 0.5,
    onChange: (e) => handleDraftChange('attributes', { ...draft.attributes, [field]: Number(e.target.value) || 0.5 }),
  })

  const activeColor = COLOR_OPTIONS.find((option) => option.value === draft.color) || COLOR_OPTIONS[0]
  const typeChoices = TYPE_OPTIONS[draft.category] || TYPE_OPTIONS.top
  const subtypeChoices = SUBTYPE_OPTIONS[draft.category] || SUBTYPE_OPTIONS.top

  return (
    <section className="space-y-6">
      <PageHeader eyebrow="Closet" title="Wardrobe overview">
        <button type="button" className="btn-secondary flex-1 sm:flex-none" onClick={onExportWardrobe}>Export</button>
        <label className="btn-secondary flex-1 sm:flex-none">
          Import
          <input type="file" accept="application/json" className="sr-only" onChange={onImportWardrobe} />
        </label>
        <button type="button" className="btn-primary w-full sm:w-auto" onClick={openNew}>+ Add clothing</button>
      </PageHeader>

      <div className="grid gap-2 rounded-2xl border border-[#e8e2dc] bg-white p-3 shadow-[0_8px_28px_rgba(42,31,35,0.04)] sm:grid-cols-[1fr_160px_160px] sm:p-4">
        <input className="input" value={search} onChange={(e) => setSearch(e.target.value)} placeholder="Search garments" />
        <select className="input" value={statusFilter} onChange={(e) => setStatusFilter(e.target.value)}>
          <option value="all">All availability</option>
          <option value="available">Available</option>
          <option value="unavailable">Unavailable</option>
        </select>
        <select className="input" value={categoryFilter} onChange={(e) => setCategoryFilter(e.target.value)}>
          <option value="all">All categories</option>
          <option value="top">Top</option>
          <option value="bottom">Bottom</option>
          <option value="outerwear">Outerwear</option>
          <option value="accessory">Accessory</option>
          <option value="shoe">Shoe</option>
        </select>
      </div>

      {isFormOpen ? (
        <div className="fixed inset-0 z-50 flex items-end justify-center overflow-y-auto bg-[#211a20]/55 p-0 backdrop-blur-[2px] sm:items-center sm:p-4" onClick={closeForm}>
          <div className="max-h-[92dvh] w-full max-w-3xl overflow-y-auto rounded-t-[26px] bg-[#fffdfa] p-4 shadow-2xl sm:rounded-[26px] sm:p-6" onClick={(e) => e.stopPropagation()}>
            <div className="mb-5 flex flex-wrap items-end justify-between gap-3">
              <div>
                <p className="text-sm font-medium text-accent">{editingId ? 'Edit clothing' : 'Add clothing'}</p>
                <h2 className="text-2xl font-bold tracking-tight">{mode === 'quick' ? 'Quick add' : 'Detailed add'}</h2>
              </div>
              <div className="flex gap-2">
                <button type="button" className={mode === 'quick' ? 'pill active' : 'pill'} onClick={() => setMode('quick')}>Quick</button>
                <button type="button" className={mode === 'detailed' ? 'pill active' : 'pill'} onClick={() => setMode('detailed')}>Detailed</button>
              </div>
            </div>

            <div className="grid gap-4 sm:grid-cols-2">
              <Field label="Garment image" className="sm:col-span-2">
                <input
                  type="file"
                  accept="image/jpeg,image/png,image/webp,image/avif"
                  onChange={(e) => setImageFile(e.target.files?.[0] || null)}
                  className="block w-full text-sm text-slate-600 file:mr-3 file:rounded-lg file:border-0 file:bg-accent-soft file:px-3 file:py-2 file:text-sm file:font-semibold file:text-accent hover:file:bg-violet-200"
                />
              </Field>

              <div className="sm:col-span-2 flex items-center gap-3 rounded-xl border border-slate-200 bg-slate-50 p-3">
                <span className="h-7 w-7 shrink-0 rounded-full border border-slate-200" style={{ background: activeColor.hex }} />
                <div>
                  <p className="text-xs font-medium uppercase tracking-[0.08em] text-slate-500">Quick summary</p>
                  <p className="text-sm font-semibold text-slate-800">{draft.name || 'Untitled garment'} · {activeColor.label}</p>
                </div>
              </div>

              <Field label="Garment name" className="sm:col-span-2"><input {...text('name')} /></Field>
              <Field label="Category">
                <select className="input" value={draft.category} onChange={(e) => {
                  const nextCategory = e.target.value
                  const nextType = TYPE_OPTIONS[nextCategory]?.[0] || 'tee'
                  const nextSubtype = SUBTYPE_OPTIONS[nextCategory]?.[0] || 'basic'
                  handleDraftChange('category', nextCategory)
                  handleDraftChange('type', nextType)
                  handleDraftChange('subtype', nextSubtype)
                }}>
                  <option value="top">Top</option>
                  <option value="bottom">Bottom</option>
                  <option value="outerwear">Outerwear</option>
                  <option value="accessory">Accessory</option>
                  <option value="shoe">Shoe</option>
                </select>
              </Field>
              <Field label="Type">
                <select className="input" value={draft.type} onChange={(e) => handleDraftChange('type', e.target.value)}>
                  {typeChoices.map((option) => <option key={option} value={option}>{option}</option>)}
                </select>
              </Field>
              <Field label="Subtype">
                <select className="input" value={draft.subtype} onChange={(e) => handleDraftChange('subtype', e.target.value)}>
                  {subtypeChoices.map((option) => <option key={option} value={option}>{option}</option>)}
                </select>
              </Field>
              <Field label="Brand"><input {...text('brand')} /></Field>
              <Field label="Size">
                <select className="input" value={draft.size || 'M'} onChange={(e) => handleDraftChange('size', e.target.value)}>
                  {SIZE_OPTIONS.map((option) => <option key={option} value={option}>{option}</option>)}
                </select>
              </Field>
              <Field label="Size system">
                <select className="input" value={draft.sizeSystem} onChange={(e) => handleDraftChange('sizeSystem', e.target.value)}>
                  <option value="us">US</option>
                  <option value="uk">UK</option>
                  <option value="eu">EU</option>
                  <option value="fr">FR</option>
                  <option value="custom">Custom</option>
                </select>
              </Field>
              <Field label="Availability">
                <select className="input" value={draft.status} onChange={(e) => handleDraftChange('status', e.target.value)}>
                  <option value="available">Available</option>
                  <option value="unavailable">Unavailable</option>
                </select>
              </Field>
              <Field label="Primary color">
                <select className="input" value={draft.color} onChange={(e) => {
                  const nextColor = e.target.value
                  handleDraftChange('color', nextColor)
                  handleDraftChange('visual_attributes', {
                    ...draft.visual_attributes,
                    primary_colors: [nextColor],
                  })
                }}>
                  {COLOR_OPTIONS.map((option) => (
                    <option key={option.value} value={option.value}>{option.label}</option>
                  ))}
                </select>
              </Field>
              <Field label="Fit">
                <select className="input" value={draft.fit} onChange={(e) => handleDraftChange('fit', e.target.value)}>
                  <option value="fitted">Fitted</option>
                  <option value="regular">Regular</option>
                  <option value="relaxed">Relaxed</option>
                  <option value="oversized">Oversized</option>
                </select>
              </Field>
              <Field label="Style">
                <select className="input" value={draft.style} onChange={(e) => handleDraftChange('style', e.target.value)}>
                  {STYLE_OPTIONS.map((option) => <option key={option} value={option}>{option}</option>)}
                </select>
              </Field>
              <Field label="Aesthetic"><input {...text('aesthetic')} /></Field>
              {mode === 'detailed' ? (
                <>
                  <Field label="Pattern"><input {...text('pattern')} /></Field>
                  <Field label="Material"><input {...text('material')} /></Field>
                  <Field label="Formality (0 to 1)"><input {...score('formality_estimate')} /></Field>
                  <Field label="Visual weight (0 to 1)"><input {...score('visual_weight_estimate')} /></Field>
                </>
              ) : null}
            </div>

            <div className="mt-6 flex justify-end gap-2">
              <button type="button" className="btn-secondary" onClick={closeForm}>Cancel</button>
              <button type="button" className="btn-primary" onClick={handleSubmit}>Save garment</button>
            </div>
          </div>
        </div>
      ) : null}

      <WardrobeGrid
        garments={visibleGarments}
        onSelect={onSelectGarment}
        onEdit={(garment) => {
          setEditingId(garment.id)
          setDraft({
            ...blankDraft,
            ...garment,
            visual_attributes: garment.visual_attributes || blankDraft.visual_attributes,
            attributes: garment.attributes || blankDraft.attributes,
            fit_and_proportions: garment.fit_and_proportions || blankDraft.fit_and_proportions,
          })
          setIsFormOpen(true)
          setMode('detailed')
        }}
        onToggleAvailability={onToggleGarmentAvailability}
        onDelete={onDeleteGarment}
      />
    </section>
  )
}

function HelpPopover({ title, text }) {
  const [isOpen, setIsOpen] = useState(false)

  return (
    <div className="group relative inline-flex">
      <button
        type="button"
        className="inline-flex h-5 w-5 items-center justify-center rounded-full border border-slate-300 bg-white text-[10px] font-bold text-slate-600 transition hover:border-violet-400 hover:text-violet-700"
        aria-label={title}
        aria-expanded={isOpen}
        aria-haspopup="dialog"
        onClick={() => setIsOpen((current) => !current)}
        onBlur={() => setIsOpen(false)}
        title={title}
      >
        ?
      </button>
      {isOpen ? (
        <div role="dialog" aria-label={title} className="absolute left-0 top-7 z-20 w-64 rounded-xl border border-slate-200 bg-white p-2 text-left text-xs leading-snug text-slate-600 shadow-lg">
          <p className="font-semibold text-slate-800">{title}</p>
          <p className="mt-1">{text}</p>
        </div>
      ) : null}
    </div>
  )
}

export default function MyProfile({ profile, onProfileChange, onClearProfile, onSaveProfile }) {

  const measurements = Array.isArray(current.measurements) ? current.measurements : []
  const updateField = (field, value) => onProfileChange(normalizeProfile({ ...current, [field]: value }))

  const updateMeasurement = (index, field, value) => {
    const next = [...measurements]
    next[index] = { ...next[index], [field]: value }
    onProfileChange(normalizeProfile({ ...current, measurements: next }))
  }

  const addMeasurement = () => {
    onProfileChange(normalizeProfile({
      ...current,
      measurements: [...measurements, { name: 'chestCircumference', value: '', unit: 'cm', notes: '', confidence: 'estimated' }],
    }))
  }

  const removeMeasurement = (index) => {
    onProfileChange(normalizeProfile({ ...current, measurements: measurements.filter((_, i) => i !== index) }))
  }

  const getGuide = (name) => measurementGuides[name] || measurementGuides.chestCircumference

  return (

        <div className="settings-card space-y-4 lg:col-span-2">
          <h3 className="text-lg font-semibold">Body proportions</h3>
          <div className="flex flex-wrap gap-2">
            {['Longer torso', 'Shorter torso', 'Longer legs', 'Shorter legs', 'Broad shoulders', 'Narrow shoulders'].map((option) => (
              <button
                key={option}
                type="button"
                className={current.proportions?.includes(option) ? 'pill active' : 'pill'}
                onClick={() => {
                  const selected = current.proportions || []
                  updateField('proportions', selected.includes(option) ? selected.filter((entry) => entry !== option) : [...selected, option])
                }}
              >
                {option}
              </button>
            ))}
          </div>
        </div>

        <div className="settings-card space-y-4 lg:col-span-2">
          <div className="flex flex-wrap items-start justify-between gap-3">
            <div>
              <h3 className="text-lg font-semibold">Body measurements</h3>
              <p className="mt-1 max-w-prose text-sm text-slate-600">
                Use a soft tape, stand tall, and keep it snug but not tight. Measure in a relaxed posture for the most wearable fit.
              </p>
            </div>
            <button type="button" className="btn-primary" onClick={addMeasurement}>Add measurement</button>
          </div>

          {measurements.length === 0 ? <p className="text-sm text-slate-500">No measurements recorded yet.</p> : null}

          <div className="grid gap-4 xl:grid-cols-2">
            {measurements.map((measurement, index) => {
              const guide = getGuide(measurement.name)
              return (
                <div key={`${measurement.name || 'measurement'}-${index}`} className="space-y-3 rounded-xl border border-slate-200 bg-slate-50 p-4">
                  <div className="flex items-center gap-3">
                    <img src={guide.image} alt={guide.label} className="h-14 w-14 shrink-0 rounded-lg bg-white object-contain p-1 ring-1 ring-slate-200" />
                    <div className="min-w-0">
                      <p className="font-semibold">{guide.label}</p>
                      <p className="text-xs leading-snug text-slate-600">{guide.tip}</p>
                    </div>
                  </div>

                  <div className="grid grid-cols-[1fr_90px_80px] gap-2">
                    <select className="input" value={measurement.name || 'chestCircumference'} onChange={(e) => updateMeasurement(index, 'name', e.target.value)}>
                      {Object.entries(measurementLabels).map(([key, label]) => <option key={key} value={key}>{label}</option>)}
                    </select>
                    <input className="input" type="number" value={measurement.value ?? ''} onChange={(e) => updateMeasurement(index, 'value', Number(e.target.value) || null)} placeholder="0" />
                    <select className="input" value={measurement.unit || 'cm'} onChange={(e) => updateMeasurement(index, 'unit', e.target.value)}>
                      <option value="cm">cm</option>
                      <option value="in">in</option>
                    </select>
                  </div>

                  <div className="grid grid-cols-[150px_1fr_auto] gap-2">
                    <input className="input" type="date" value={measurement.date || ''} onChange={(e) => updateMeasurement(index, 'date', e.target.value)} />
                    <input className="input" type="text" value={measurement.notes || ''} onChange={(e) => updateMeasurement(index, 'notes', e.target.value)} placeholder="Notes" />
                    <button type="button" className="btn-secondary" onClick={() => removeMeasurement(index)}>Remove</button>
                  </div>
                </div>
              )
            })}
          </div>
        </div>
      </div>
    </section>
  )
}

import { useState } from 'react'
import { aestheticOptions } from '../data/aesthetics.js'

const palette = [
    ['ivory', '#f4f1e8'], ['white', '#ffffff'], ['black', '#17161d'], ['grey', '#8b8e94'],
    ['navy', '#1f2f4a'], ['blue', '#4a7fc1'], ['beige', '#cdb99a'], ['brown', '#6b4a32'],
    ['olive', '#6b7340'], ['green', '#2f6b4f'], ['burgundy', '#7a2236'], ['yellow', '#e8c547'],
]

const formalityLabels = ['Very relaxed', 'Easygoing', 'Balanced', 'Polished', 'Dressy']

export default function Onboarding({ name, initialPreferences, initialAesthetic, onFinish, onSkip }) {
    const [step, setStep] = useState(0)
    const [aesthetic, setAesthetic] = useState(initialAesthetic || aestheticOptions[0]?.id)
    const [colors, setColors] = useState(initialPreferences?.colors || [])
    const [formality, setFormality] = useState(initialPreferences?.formalityBias || 3)
    const total = 3

    const toggleColor = (color) => {
        setColors((current) => (
            current.includes(color) ? current.filter((entry) => entry !== color) : [...current, color]
        ))
    }

    const finish = () => {
        onFinish({
            aesthetic,
            preferences: {
                ...initialPreferences,
                styles: [aesthetic],
                colors: colors.length ? colors : initialPreferences?.colors,
                formalityBias: formality,
            },
        })
    }

    return (
        <div className="flex min-h-dvh items-center justify-center bg-[#f8f6f2] p-3 sm:p-5">
            <div className="w-full max-w-xl space-y-5 rounded-[26px] border border-[#e8e2dc] bg-[#fffdfa] p-5 shadow-[0_16px_45px_rgba(42,31,35,0.07)] sm:space-y-6 sm:p-8">
                <div className="space-y-3">
                    <div className="flex items-center justify-between text-sm">
                        <p className="font-semibold text-accent">Step {step + 1} of {total}</p>
                        <button type="button" className="min-h-10 px-2 text-slate-500 hover:text-slate-800" onClick={onSkip}>Skip for now</button>
                    </div>
                    <div className="h-1.5 overflow-hidden rounded-full bg-slate-100" role="progressbar" aria-valuemin={1} aria-valuemax={total} aria-valuenow={step + 1}>
                        <div className="h-full rounded-full bg-accent transition-all" style={{ width: `${((step + 1) / total) * 100}%` }} />
                    </div>
                </div>

                {step === 0 ? (
                    <section className="space-y-4">
                        <div>
                            <div className="flex items-center gap-2">
                                <h2 className="text-xl font-semibold leading-snug tracking-tight text-ink sm:text-2xl">{name ? `Nice to meet you, ${name}.` : 'Nice to meet you.'} What’s your style?</h2>
                                <button
                                    type="button"
                                    className="inline-flex h-5 w-5 items-center justify-center rounded-full border border-slate-300 bg-white text-[10px] font-bold text-slate-600"
                                    aria-label="Style helps the app suggest outfits that feel like you."
                                    title="Style helps the app suggest outfits that feel like you."
                                >
                                    ?
                                </button>
                            </div>
                            <p className="mt-1 text-slate-600">Pick the look you reach for most. Outfits that match it get a boost.</p>
                        </div>
                        <div className="flex flex-wrap gap-2">
                            {aestheticOptions.map((option) => (
                                <button
                                    key={option.id}
                                    type="button"
                                    aria-pressed={aesthetic === option.id}
                                    className={aesthetic === option.id ? 'pill active' : 'pill'}
                                    onClick={() => setAesthetic(option.id)}
                                >
                                    {option.label}
                                </button>
                            ))}
                        </div>
                    </section>
                ) : null}

                {step === 1 ? (
                    <section className="space-y-4">
                        <div>
                            <h2 className="text-xl font-semibold tracking-tight text-ink sm:text-2xl">Which colors do you love wearing?</h2>
                            <p className="mt-1 text-slate-600">Choose as many as you like.</p>
                        </div>
                            <div className="grid grid-cols-2 gap-2 min-[390px]:grid-cols-3 sm:grid-cols-4">
                            {palette.map(([label, hex]) => {
                                const active = colors.includes(label)
                                return (
                                    <button
                                        key={label}
                                        type="button"
                                        aria-pressed={active}
                                        onClick={() => toggleColor(label)}
                                        className={`flex min-h-11 items-center gap-2 rounded-xl border p-2.5 text-sm font-medium capitalize transition ${active ? 'border-accent bg-accent-soft' : 'border-[#e8e2dc] hover:bg-slate-50'}`}
                                    >
                                        <span className="h-5 w-5 shrink-0 rounded-full ring-1 ring-slate-300" style={{ background: hex }} />
                                        {label}
                                    </button>
                                )
                            })}
                        </div>
                    </section>
                ) : null}

                {step === 2 ? (
                    <section className="space-y-4">
                        <div>
                            <div className="flex items-center gap-2">
                                <h2 className="text-xl font-semibold tracking-tight text-ink sm:text-2xl">How dressed up do you like to be?</h2>
                                <button
                                    type="button"
                                    className="inline-flex h-5 w-5 items-center justify-center rounded-full border border-slate-300 bg-white text-[10px] font-bold text-slate-600"
                                    aria-label="Formality bias sets your default polish level for outfit suggestions."
                                    title="Formality bias sets your default polish level for outfit suggestions."
                                >
                                    ?
                                </button>
                            </div>
                            <p className="mt-1 text-slate-600">This is your everyday default. Context changes it for events.</p>
                        </div>
                        <input type="range" min="1" max="5" value={formality} onChange={(event) => setFormality(Number(event.target.value))} className="w-full accent-violet-600" aria-label="Formality bias" />
                        <div className="flex justify-between text-xs text-slate-500"><span>Relaxed</span><span>Dressy</span></div>
                        <p className="text-center text-lg font-semibold">{formalityLabels[formality - 1]}</p>
                    </section>
                ) : null}

                <div className="sticky bottom-0 -mx-5 -mb-5 flex items-center justify-between border-t border-[#eee8e2] bg-[#fffdfa]/95 px-5 py-4 backdrop-blur sm:static sm:mx-0 sm:mb-0 sm:border-0 sm:bg-transparent sm:px-0 sm:pb-0">
                    <button type="button" className="btn-secondary" disabled={step === 0} onClick={() => setStep((current) => current - 1)}>Back</button>
                    {step < total - 1 ? (
                        <button type="button" className="btn-primary" onClick={() => setStep((current) => current + 1)}>Next</button>
                    ) : (
                        <button type="button" className="btn-primary" onClick={finish}>Finish setup</button>
                    )}
                </div>
            </div>
        </div>
    )
}
